program main{
	var a, b, i, r : int;
	begin;
		write("prueba operadores");

		a := 17;
		b := 5;

		write(a + b);
		write(a - b);
		write(a * b);
		write(a / b);
		write(a % b);
		write(-a);
		write(-a + b);
		write(a * -b);

		if (a >  b)  then { write(1); } else { write(0); }
		if (a <  b)  then { write(1); } else { write(0); }
		if (a >= a)  then { write(1); } else { write(0); }
		if (a <= b)  then { write(1); } else { write(0); }
		if (a == 17) then { write(1); } else { write(0); }
		if (a != b)  then { write(1); } else { write(0); }

		if (a > b and b > 0) then { write(1); } else { write(0); }
		if (a < b or  b > 0) then { write(1); } else { write(0); }
		if (not (a < b))     then { write(1); } else { write(0); }

		r := 0;
		while (r <= 3) do { write(r); r := r + 1; }

		for (i := 3; i >= 1; i--) { write(i); }
	end;
}
