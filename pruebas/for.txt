program main{
	var i, iteraciones, actual, anterior, temporal : int;
	begin;
		write("--- SERIE FIBONACCI CON FOR ---");
		iteraciones := 8;
		anterior := 0;
		actual := 1;
		
		write(anterior);
		write(actual);

		for (i := 2; i < iteraciones; i++) {
			temporal := actual;
			actual := actual + anterior;
			anterior := temporal;
			write(actual);
		}
	end;
}
